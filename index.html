Acesso à conta

Configuração necessária no Supabase

Crie um bucket privado chamado pediatria-pdfs e execute o SQL abaixo no SQL Editor. As políticas restringem os dados ao usuário autenticado.

create table if not exists public.ped_docs (
 id uuid primary key,
 user_id uuid not null references auth.users(id) on delete cascade,
 name text not null,
 page_count integer not null default 0,
 extracted_pages integer not null default 0,
 pending_ocr integer not null default 0,
 storage_path text,
 created_at timestamptz default now()
);
create table if not exists public.ped_pages (
 id uuid primary key,
 user_id uuid not null references auth.users(id) on delete cascade,
 doc_id uuid not null references public.ped_docs(id) on delete cascade,
 page_no integer not null,
 text_content text not null default '',
 has_text boolean not null default false,
 unique(doc_id,page_no)
);
create table if not exists public.ped_notes (
 id uuid primary key,
 user_id uuid not null references auth.users(id) on delete cascade,
 title text not null,
 body text not null,
 updated_at timestamptz default now()
);
create table if not exists public.ped_cards (
 id uuid primary key,
 user_id uuid not null references auth.users(id) on delete cascade,
 topic text not null default 'Pediatria',
 front text not null,
 back text not null,
 source text,
 due_at timestamptz,
 interval_days integer default 0,
 ease numeric default 2.5,
 created_at timestamptz default now()
);
alter table public.ped_docs enable row level security;
alter table public.ped_pages enable row level security;
alter table public.ped_cards enable row level security;
alter table public.ped_notes enable row level security;
create policy "own docs" on public.ped_docs for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own pages" on public.ped_pages for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own cards" on public.ped_cards for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own notes" on public.ped_notes for all using (auth.uid()=user_id) with check (auth.uid()=user_id);

insert into storage.buckets (id,name,public)
values ('pediatria-pdfs','pediatria-pdfs',false)
on conflict (id) do nothing;
create policy "Pediatria PDF select own" on storage.objects for select
using (bucket_id='pediatria-pdfs' and (storage.foldername(name))[1]=auth.uid()::text);
create policy "Pediatria PDF insert own" on storage.objects for insert
with check (bucket_id='pediatria-pdfs' and (storage.foldername(name))[1]=auth.uid()::text);
create policy "Pediatria PDF update own" on storage.objects for update
using (bucket_id='pediatria-pdfs' and (storage.foldername(name))[1]=auth.uid()::text)
with check (bucket_id='pediatria-pdfs' and (storage.foldername(name))[1]=auth.uid()::text);
create policy "Pediatria PDF delete own" on storage.objects for delete
using (bucket_id='pediatria-pdfs' and (storage.foldername(name))[1]=auth.uid()::text);
A conexão com nuvem só fica operacional depois de criar o projeto, bucket, tabelas e políticas. Até lá, os dados permanecem apenas neste navegador. Não use service_role key no HTML.
